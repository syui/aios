[[ abc == a* ]] && echo match	match
[[ abc == "a*" ]] || echo quoted	quoted

echo }	}

p=(a b c); echo ${#p} ${p[2]} ${p[-1]}	3 b c
setopt ksharrays; p=(a b c); echo ${p[0]}	a
p=(a); p+=(b c); echo ${p[@]}	a b c

setopt nullglob; echo x nomatch*.zz y	x y
